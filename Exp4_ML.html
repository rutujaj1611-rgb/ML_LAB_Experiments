STEP 8: EXPLAINED VARIANCE¶

print("\n========== EXPLAINED VARIANCE ==========")

pc1_variance = pca.explained_variance_ratio_[0]

pc2_variance = pca.explained_variance_ratio_[1]

total_variance = ( pc1_variance + pc2_variance )

print( "PC1 variance:", pc1_variance * 100, "%" )

print( "PC2 variance:", pc2_variance * 100, "%" )

print( "Total variance retained:", total_variance * 100, "%" )
